class Tier1AdaptiveTrigger:
    """
    Candidate Patent A: Zone-Aware Adaptive Two-Tier Edge Inference.
    Runs low-compute background subtraction or frame difference.
    Wakes up heavy quantized model only on motion threshold breach.
    """
    def __init__(self, sensitivity_threshold=1500):
        self.subtractor = cv2.createBackgroundSubtractorMOG2(history=100, varThreshold=50, detectShadows=False)
        self.sensitivity = sensitivity_threshold

    def evaluate_zone_activity(self, frame, roi_coords) -> bool:
        x, y, w, h = roi_coords
        roi = frame[y:y+h, x:x+w]
        fg_mask = self.subtractor.apply(roi)
        motion_energy = np.sum(fg_mask > 0)
        return motion_energy > self.sensitivity

class PredictiveQueueEngine:
    """
    Candidate Patent C: Predictive Queue Congestion Index (PQCI)
    Computes Temporal Flow Imbalance Factor (TFIF = Flow_in / Flow_bill)
    """
    def __init__(self, window_sec=60):
        self.inflow_history = []
        self.billing_history = []
        self.window = window_sec

    def update_rates(self, entries_delta: int, billed_delta: int):
        now = time.time()
        self.inflow_history.append((now, entries_delta))
        self.billing_history.append((now, billed_delta))
        self._prune(now)

    def _prune(self, now):
        self.inflow_history = [x for x in self.inflow_history if now - x[0] <= self.window]
        self.billing_history = [x for x in self.billing_history if now - x[0] <= self.window]

    def compute_pqci(self) -> Dict:
        in_sum = sum(x[1] for x in self.inflow_history) or 1
        bill_sum = sum(x[1] for x in self.billing_history) or 1
        tfif = round(in_sum / bill_sum, 2)

        congested = tfif > TFIF_CONGESTION_THRESHOLD
        eta_minutes = round(max(1.5, 7.0 - tfif), 1) if congested else None

        return {
            "tfif": tfif,
            "congested": congested,
            "predicted_congestion_eta_min": eta_minutes,
            "action": "OPEN_COUNTER_2" if congested else "STANDBY"
        }

class InventoryDensityEngine:
    """
    Edge-density texture variance for shelf out-of-stock estimation.
    Avoids expensive multi-hundred SKU object detection models.
    """
    @staticmethod
    def calculate_stock_density(shelf_roi) -> float:
        gray = cv2.cvtColor(shelf_roi, cv2.COLOR_BGR2GRAY)
        edges = cv2.Canny(gray, 50, 150)
        density = np.count_nonzero(edges) / edges.size
        return density
